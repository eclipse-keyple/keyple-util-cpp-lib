81bool
82HexUtil::isValid(const std::string& hex)
83{
84 if (hex.length() == 0 || hex.length() % 2 != 0) {
85 return false;
86 }
87
88 for (int i = 0; i < static_cast<int>(hex.size()); i++) {
89 if (mHexToNibble[hex.at(i)] == 0xFF) {
90 return false;
91 }
92 }
93
94 return true;
95}

97const std::vector<uint8_t>
98HexUtil::toByteArray(const std::string& hex)
99{
100 std::vector<uint8_t> tab(hex.size() / 2);
101
102 if (hex.length() % 2) {
103 throw StringIndexOutOfBoundsException("string has odd length");
104 }
105
106 for (int i = 0; i < static_cast<int>(hex.length()); i += 2) {
107 tab[i / 2]
108 = ((mHexToNibble[hex.at(i)] << 4)
109 + (mHexToNibble[hex.at(i + 1)] & 0xFF));
110 }
111
112 return tab;
113}

195const std::string
196HexUtil::toHex(const uint32_t val)
197{

203 return mByteToHex[val >> 16 & 0xFF] + mByteToHex[val >> 8 & 0xFF]
